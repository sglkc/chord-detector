\title[Supplementary material: Evaluating CNN vs.\ SVM for Piano Chord Recognition]{Supplementary material: Evaluating CNN vs.\ SVM for Piano Chord Recognition Under Acoustic Domain Shift}

\author*[1]{\fnm{Cikal} \sur{Gemintang Seya}}\email{cikalgemintangseya1@gmail.com}

\author[1]{\fnm{Jasman} \sur{Pardede}}\email{jasman@itenas.ac.id}

\affil*[1]{\orgdiv{Informatics}, \orgname{National Institute of Technology}, \orgaddress{\street{Neglasari}, \city{Bandung}, \postcode{40124}, \state{West Java}, \country{Indonesia}}}

\abstract{This supplementary material provides additional figures and tables for the main article on Constant-Q Transform (CQT) spectrogram features with CNN and RBF-SVM classifiers under acoustic domain shift. Included are dataset acquisition and duration diagnostics, CNN training curves and in-domain confusion matrices, full 5-fold cross-validation metrics for both classifiers, SVM hyperparameter search results, and a comparative performance summary plot.}
